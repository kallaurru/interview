Дан массив целых неотрицательных чисел, нужно сгруппировать друг с другом числа,
которые можно получить путем перестановки цифр их составляющих, нули при этом игно
рируются, т.к. нет числа 011.

expect = []int{1230, 99, 23001, 123, 111, 300021, 101010, 900009, 9}
actual = [][]int {{99, 900009}, {1230, 123, 23001, 300021}, {111, 101010}, {9}}

expect = []int{11, 22}
actual = [][]int {{11}, {22}}
